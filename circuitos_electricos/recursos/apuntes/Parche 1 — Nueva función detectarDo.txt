/* =====================================================================
   4b) Detección de topología Doble T
   Patrón buscado:
     - 5 nodos en total
     - Nodos '1' y '2' presentes (terminales)
     - Dos "centros" (o1, o2), cada uno conectado a '1' y a '2'
     - Ambos centros conectados entre sí
     - Un quinto nodo conectado a exactamente uno de los centros (tierra)
   Retorna { o1, o2, ground } o null si no coincide.
   ===================================================================== */
function detectarDobleT(elementos, nodos) {
  if (nodos.length !== 5) return null;
  const nStr = nodos.map(String);
  if (!nStr.includes('1') || !nStr.includes('2')) return null;

  const adj = {};
  nodos.forEach(n => adj[n] = new Set());
  elementos.forEach(el => { adj[el.a].add(el.b); adj[el.b].add(el.a); });

  const candidatos = nodos.filter(n =>
    n !== '1' && n !== '2' && adj[n].has('1') && adj[n].has('2')
  );
  if (candidatos.length !== 2) return null;

  const [c1, c2] = candidatos;
  if (!adj[c1].has(c2)) return null;

  const quinto = nodos.find(n => n !== '1' && n !== '2' && n !== c1 && n !== c2);
  if (!quinto) return null;

  const connC1 = adj[quinto].has(c1);
  const connC2 = adj[quinto].has(c2);
  if (connC1 === connC2) return null;

  const o2 = connC1 ? c1 : c2;
  const o1 = o2 === c1 ? c2 : c1;
  return { o1, o2, ground: quinto };
}
